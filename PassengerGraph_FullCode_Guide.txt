private void BuildSeries(List<PassengerCountLineGraphPoint> points)
{
    TrainNames.Clear();
    SeriesMap.Clear();

    var palette = GetPalette();

    var groups = points
        .GroupBy(x => new { x.TrainNo, x.TripId })
        .OrderBy(g => g.Min(x => x.DDateTime))
        .ThenBy(g => g.Key.TrainNo)
        .ThenBy(g => g.Key.TripId)
        .ToList();

    var series = new List<ISeries>();

    LiveChartsCore.LiveCharts.Configure(config =>
        config.HasMap<PassengerCountLineChartPoint>(
            (point, index) =>
                new LiveChartsCore.Kernel.Coordinate(
                    point.X ?? 0,
                    point.Y ?? 0))
    );

    int colorIndex = 0;

    foreach (var group in groups)
    {
        string train = group.Key.TrainNo;
        int tripId = group.Key.TripId;
        string seriesKey = train + "_" + tripId;

        if (!TrainNames.Contains(train))
            TrainNames.Add(train);

        var ordered = group
            .OrderBy(x => x.StationIndex)
            .ToList();

        var values = ordered.Select(x =>
            new PassengerCountLineChartPoint
            {
                X = x.StationIndex - 1,
                Y = x.TotalPassenger,

                TrainNo = train,
                TripId = tripId,

                DDateTime = x.DDateTime,

                CurrentStation = x.CurrentStation,
                NextStation = x.NextStation,

                TotalPassenger = x.TotalPassenger
            }).ToArray();

        SKColor color =
            palette[colorIndex % palette.Count];

        colorIndex++;

        var line =
            new LineSeries<PassengerCountLineChartPoint>
            {
                Name = seriesKey,
                Values = values,

                GeometrySize = 7,
                LineSmoothness = 0,
                Fill = null,

                Stroke = new SolidColorPaint(color)
                {
                    StrokeThickness = 3
                },

                XToolTipLabelFormatter = chartPoint =>
                {
                    var data =
                        (PassengerCountLineChartPoint)
                        chartPoint.Model;

                    return
                        $"Train: {data.TrainNo}  Trip: {data.TripId}
" +
                        $"Time: {data.DDateTime:yyyy-MM-dd HH:mm:ss.fff}
" +
                        $"{data.CurrentStation} → {data.NextStation}
" +
                        $"Passengers: {data.TotalPassenger}";
                }
            };

        SeriesMap[seriesKey] = line;

        series.Add(line);
    }

    Series = series.ToArray();
}

FILE 10: Station ordering rule
========================================================

Do NOT use StationCode to determine graph order.

Use:

Corridor + Direction + StationIndex

from:

CorridorDirectionStations

Example Corridor 3 Direction 1:

1  Rai Durg
2  HITEC City
3  Durgam Cheruvu
4  Madhapur
...
23 Secunderabad

Therefore:

X = StationIndex - 1

is correct.
